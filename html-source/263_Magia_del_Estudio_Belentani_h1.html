BACKEND (Python + FastAPI) #### 📄 `requirements.txt` ```text faster-whisper librosa numpy fastapi uvicorn python-multipart ``` *(Instalar con: `pip install -r requirements.txt`)* #### 📄 `whisper_engine.py` (El Oído que todo lo escucha) Usamos `faster-whisper` para que corra en tu CPU/GPU local. Nadie escucha tus confesiones excepto tú. ```python from faster_whisper import WhisperModel import torch class WhisperEngine: def __init__(self, model_size="tiny"): # "tiny" para velocidad, "base" para precisión device = "cuda" if torch.cuda.is_available() else "cpu" compute_type = "float16" if device == "cuda" else "int8" print(f"🎙️ Iniciando Whisper Local en {device.upper()}...") self.model = WhisperModel(model_size, device=device, compute_type=compute_type) def transcribe(self, audio_path): segments, info = self.model.transcribe(audio_path, beam_size=5, language="es") text = " ".join([segment.text for segment in segments]) return text.strip() ``` #### 📄 `audio_analyzer.py` (El Tercer Oído) Aquí ocurre la magia científica. Extraemos la energía (para saber si gritas o susurras), el tono (emoción) y los silencios (tu ritmo respiratorio). ```python import librosa import numpy as np class AudioAnalyzer: def analyze(self, audio_path): y, sr = librosa.load(audio_path, sr=None) # 1. ENERGÍA (¿Susurro o Grito?) rms = librosa.feature.rms(y=y)[0] avg_energy = np.mean(rms) # 2. TONO (¿Emoción?) pitches, magnitudes = librosa.piptrack(y=y, sr=sr) index = magnitudes.argmax(axis=0) pitch_vals = [pitches[index[t], t] for t in range(pitches.shape[1]) if pitches[index[t], t] > 0] avg_pitch = np.mean(pitch_vals) if pitch_vals else 150 # 3. RESPIRACIÓN (¿Pausas/Silencios?) intervals = librosa.effects.split(y, top_db=20) pause_durations = [] for i in range(len(intervals)-1): pause = intervals[i+1][0] - intervals[i][1] pause_durations.append(pause / sr) avg_pause = np.mean(pause_durations) if pause_durations else 0 return self._interpret(avg_energy, avg_pitch, avg_pause) def _interpret(self, energy, pitch, pause): intention = "neutral" emotion = "calm" # Detectar INTENCIÓN MÁGICA if energy < 0.01: intention = "whisper" elif energy > 0.15: intention = "shout" elif pause > 1.5: intention = "confess" # Pausas largas = confesión elif pitch > 250 and energy < 0.05: intention = "pray" # Detectar EMOCIÓN if pitch > 300 and energy > 0.1: emotion = "anxious" elif pitch < 150 and energy < 0.05: emotion = "sad" elif energy > 0.12: emotion = "passionate" return { "intention": intention, "emotion": emotion, "metrics": { "energy": float(energy), "pitch": float(pitch), "breathing_rhythm": float(pause) } } ``` #### 📄 `intention_router.py` (El Guía de Oráculos) ```python class IntentionRouter: def __init__(self): self.oracles = { "dream": ["soñé", "sueño", "dormir", "noche", "vision", "serpiente"], "soul": ["quién soy", "arquetipo", "siento que", "espejo", "identidad"], "karma": ["traición", "deuda", "judas", "perdón", "culpa", "robo"], "frequency": ["vibración", "frecuencia", "sonido", "color", "energía"], "timeline": ["futuro", "destino", "camino", "siguiente", "predicción"] } def route(self, text, intention, emotion): text_lower = text.lower() # Prioridad por intención mágica if intention == "confess": return "karma" if intention == "pray": return "frequency" # Búsqueda de keywords for oracle, keywords in self.oracles.items(): if any(kw in text_lower for kw in keywords): return oracle # Fallback por emoción if emotion == "anxious": return "frequency" if emotion == "sad": return "soul" return "soul" # Default: Soul Mirror ``` #### 📄 `main.py` (El Servidor Mágico) ```python from fastapi import FastAPI, UploadFile, File from fastapi.middleware.cors import CORSMiddleware import shutil, os from whisper_engine import WhisperEngine from audio_analyzer import AudioAnalyzer from intention_router import IntentionRouter app = FastAPI(title="Judas Voice Oracle API") app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"]) whisper = WhisperEngine(model_size="tiny") analyzer = AudioAnalyzer() router = IntentionRouter() @app.post("/api/v1/voice/oracle") async def voice_oracle(audio: UploadFile = File(...)): temp_path = f"temp_voice_{audio.filename}" with open(temp_path, "wb") as buffer: shutil.copyfileobj(audio.file, buffer) try: text = whisper.transcribe(temp_path) if not text: return {"error": "El silencio es pesado. Habla, agente."} audio_metrics = analyzer.analyze(temp_path) oracle_id = router.route(text, audio_metrics['intention'], audio_metrics['emotion']) # Aquí conectarías con tu Tiny Ollama RAG para generar la respuesta profunda response_text = f"El {oracle_id.upper()} ha escuchado tu {audio_metrics['intention']}. Procesando tu verdad..." return { "transcription": text, "oracle": oracle_id, "intention": audio_metrics['intention'], "emotion": audio_metrics['emotion'], "metrics": audio_metrics['metrics'], "response": response_text } finally: if os.path.exists(temp_path): os.remove(temp_path) # Ejecutar con: uvicorn main:app --reload --port 8000 ``` --- ### 🎨 2.
